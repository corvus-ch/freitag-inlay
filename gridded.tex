\documentclass[twoside]{scrartcl}

\usepackage{inlay}
\usepackage{tikz}
\usepackage{tikzpagenodes}

\usetikzlibrary{calc}

\newcommand{\griddedpage}{%
  \begin{tikzpicture}[
    remember picture,
    overlay,
    help lines
  ]
    \draw
      [
        shift={($(current page text area.north west)-(4mm, 0mm)$)},
        xstep=10mm,
        ystep=4mm
      ]
      ($(current page text area.south west)-(4mm, 0mm)$)
      grid
      ($(current page text area.north east)+(4mm, 0mm)$);

    \draw
      [dashed]
      (current page text area.south west)
      --
      (current page text area.north west);
  \end{tikzpicture}
  \clearpage
}

\begin{document}
\inlaytitlepage{Gridded}
\griddedpage
\griddedpage
\griddedpage
\griddedpage
\end{document}
