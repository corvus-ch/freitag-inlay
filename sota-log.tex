\newcommand{\reportline}[1][]{
    \node
      #1
      [
        anchor=south west,
        color=black!20,
        inner xsep=4mm,
        inner ysep=0,
        minimum height=7.5mm,
        on chain,
        outer sep=0,
        text width=\the\dimexpr \textwidth - 8mm \relax,
      ]
      {\Large\sffamily\hspace{5mm}:\hfill599\hspace{1em}599};
}

\newdimen\bodyheight

\newcommand{\sotalogpage}{%
  \begin{tikzpicture}[
    ampersand replacement=\&,
    help lines,
    node distance=7.5mm,
    overlay,
    remember picture,
    start chain=going above,
  ]

    \node
      (title)
      at (current page text area.north west)
      [
        anchor=north west,
        color=black!20,
        inner xsep=0,
        rectangle split parts=2,
        rectangle split,
        text width=\textwidth,
      ]
      {
        \Huge
        \sffamily
        \the\year{}\textcolor{black}{-}01\textcolor{black}{-}01
        \hfill
        1X\textcolor{black}{/}AA\textcolor{black}{-}000
        \nodepart{second}
        \Large
        \sffamily
        % TODO why is the text color bleading over?
        %0000.0000 \textcolor{black}{MHz}
        0000.0000 MHz
        \hfill
        SSB
        \hfill
        1000 \textcolor{black}{W}
      };

    \reportline[(rpt) at ($(current page text area.south west)+(0,7.5mm)$)]

    \tikzmath {
      coordinate \bodyheightcoord, \reportlineheightcoord;
      \bodyheightcoord = (title.south west) - (current page text area.south west);
      \reportlineheightcoord = (rpt.north) - (rpt.south);
      \bodyheightmax = veclen(\bodyheightcoord);
      \reportlineheight = veclen(\reportlineheightcoord) * 2;
      \rn = floor(\bodyheightmax / \reportlineheight) - 2;
      \bodyheight = \reportlineheight * (\rn + 2);
    };

    \foreach \r in {0,...,\rn} {\reportline}

    \draw
      [
        loosely dotted,
        shift={($(current page text area.south west)+(17.5mm,7.5mm)$)},
        color=black!60,
        semithick,
        xstep=\textwidth,
        ystep=15mm,
      ]
      ($(current page text area.south west)+(17.5mm,0)$)
      grid
      (\textwidth - 17.5mm, \bodyheight - 7.5mm);

    \draw
      [
        loosely dashed,
        shift={(current page text area.south west)},
        xstep=\textwidth,
        ystep=15mm,
      ]
      (current page text area.south west)
      grid
      (\textwidth, \bodyheight);
  \end{tikzpicture}
  \clearpage
}
