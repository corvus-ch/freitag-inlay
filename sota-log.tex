\newcommand{\reportline}[1][]{
    \node
      #1
      [
        anchor=south west,
        color=black!10,
        inner xsep=4mm,
        inner ysep=0,
        minimum height=7.5mm,
        on chain,
        outer sep=0,
        text width=\the\dimexpr \textwidth - 8mm \relax,
      ]
      {\Large\sffamily\hspace{5mm}:\hfill599\hspace{1em}599};
}

\newcommand{\sotalogpage}{%
  \begin{tikzpicture}[
    ampersand replacement=\&,
    help lines,
    node distance=7.5mm,
    overlay,
    remember picture,
    start chain=going above,
  ]

    \node
      (title)
      at (current page text area.north west)
      [
        anchor=north west,
        color=black!10,
        inner xsep=0,
        rectangle split parts=2,
        rectangle split,
        text width=\textwidth,
      ]
      {
        \Huge
        \sffamily
        \the\year{}\textcolor{black}{-}01\textcolor{black}{-}01
        \hfill
        1X\textcolor{black}{/}AA\textcolor{black}{-}000
        \nodepart{second}
        \Large
        \sffamily
        % TODO why is the text color bleading over?
        %0000.0000 \textcolor{black}{MHz}
        0000.0000 MHz
        \hfill
        SSB
        \hfill
        1000 \textcolor{black}{W}
      };

    \reportline[at ($(current page text area.south west)+(0,7.5mm)$)]
    \foreach \r in {0,...,8} {\reportline}

    \draw
      [
        loosely dotted,
        shift={($(current page text area.south west)+(17.5mm,0)$)},
        xstep=\textwidth,
        ystep=7.5mm,
      ]
      ($(current page text area.south west)+(17.5mm,0)$)
      grid
      (title.south east);

    \draw
      [
        loosely dashed,
        shift={(current page text area.south west)},
        xstep=\textwidth,
        ystep=15mm,
      ]
      (current page text area.south west)
      grid
      (title.south east);
  \end{tikzpicture}
  \clearpage
}
