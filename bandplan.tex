\newenvironment{band}[2]{\noindent
  \begin{minipage}[t]{7em}
	\vspace{0pt}
  \begin{tikzpicture}
  \renewcommand{\bandlower}{#1}
	\tikzmath{
		\bandspan = \fpeval{#2 - #1};
		\bandscale = \bandheight / \bandspan;
	}
	\freq[start]{#1}
	\freq[end]{#2}
}
{\end{tikzpicture}\end{minipage}}
